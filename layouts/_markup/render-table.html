{{- /* Markdown table render hook. Every table gets a labelled, keyboard- scrollable wrapper. Tables of 2-3 columns also get class "table--stack": under 480px each row becomes a card -- the first cell as its title, later cells labelled with their column header (data-label). 4+ columns scroll sideways instead, with an edge shadow as the hint. */ -}} {{- $cols := 0 }}{{ with .THead }}{{ with index . 0 }}{{ $cols = len . }}{{ end }}{{ end }} {{- $headers := slice }} {{- with .THead }}{{ range index . 0 }}{{ $headers = $headers | append (.Text | plainify | htmlUnescape) }}{{ end }}{{ end }} {{- $label := delimit $headers ", " }} {{- $stack := and (ge $cols 2) (le $cols 3) }}
{{- range .THead }} {{- range . }} {{ .Text }} {{- end }} {{- end }} {{- range .TBody }} {{- range $i, $c := . }} {{ $c.Text }} {{- end }} {{- end }}
